def getPeaks(filename):
  peak_list = [];
  file = open(filename,'r');
  file.readline();
  for line in file:
    parts = line.split(',');
    frequency = float(parts[0]);
    decay_constant = float(parts[1]);
    Q = float(parts[2]);
    amplitude = float(parts[3]);
    phase = float(parts[4]);
    error = float(parts[5]);
    peak_list.append([frequency, decay_constant, Q, amplitude, phase, error])
  file.close();
  return peak_list;
